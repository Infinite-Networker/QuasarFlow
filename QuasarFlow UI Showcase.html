Button Component Usage

Drop these snippets into your QuasarFlow project to summon buttons from the cosmos.

  Quantum Button Components.js
// ── Create a Nova CTA button ────────────────────────── let btn = NovaButton("Ignite the Stars", "lg"); print(btn); // ── Event Horizon (destructive) button ──────────────── let dangerBtn = EventHorizonButton("Collapse Universe", "md"); print(dangerBtn); // ── Pulsar animated button ───────────────────────────── let pulseBtn = PulsarButton("Pulse Signal", "sm"); print(pulseBtn); // ── Button Group cluster ─────────────────────────────── let cluster = [ NovaButton("Primary", "md"), PulsarButton("Secondary", "md"), GhostButton("Cancel", "md") ]; let group = ButtonGroup(); group.orientation = "horizontal"; print(createButtonGroup(cluster, group));

// ── Load fonts into the quantum registry ────────────── loadFont(QuasarSans); loadFont(PulsarDisplay); loadFont(WarpMono); // ── Render a supernova hero headline ────────────────── print(HeroText("QuasarFlow")); // ── Section heading ──────────────────────────────────── print(HeadingText("Quantum-Warp Constructs")); // ── Body paragraph ───────────────────────────────────── print(BodyText("Where code resonates with distant quasars.")); // ── WarpMono code snippet ────────────────────────────── print(CodeText("let ptr = entangle_memory(\"cosmic_sector_42\");")); // ── Cosmic quote callout ─────────────────────────────── print(QuoteText("The cosmos is not silent — it speaks in code. 🌌"));
